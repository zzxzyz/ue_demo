import "UnityEngine"
import "UnityEngine.UI"

-- 加载多语言模块
local I18n = require("i18n")

-- 根据菜单设置决定是否加载 LuaPanda 调试器
-- 只有在"选项/LuaPanda"菜单项勾选时才启动调试器
local luaPandaEnabled = _G._LuaPandaEnabled
if luaPandaEnabled then
    local ok, luaPanda = pcall(function() return require("LuaPanda") end)
    if ok and luaPanda then
        luaPanda.start("127.0.0.1", 8818)
        Debug.Log("[LoginManager] LuaPanda 调试器已启动")
    else
        Debug.LogWarning("[LoginManager] LuaPanda 调试器未找到，跳过调试器初始化")
    end
else
    Debug.Log("LuaPanda 调试器未启用（请在Unity Editor菜单：选项/LuaPanda 中启用）")
end

-- LoginManager Lua 实现
-- 实现与 LoginManager.cs 相同的功能

local LoginManager = {}

-- ========== 配置变量 ==========
-- 控制使用同步还是异步登录方式
-- true: 使用异步登录（带回调）
-- false: 使用同步登录（直接返回结果）
-- 从C#传递的全局变量读取设置（如果未设置则默认使用异步登录）
if _G._AsyncLoginEnabled ~= nil then
    LoginManager.UseAsyncLogin = _G._AsyncLoginEnabled
else
    LoginManager.UseAsyncLogin = true
end

-- 存储UI组件引用
local usernameInput = nil
local passwordInput = nil
local usernameText = nil
local passwordText = nil
local loginButton = nil
local cancelButton = nil
local settingsButton = nil
local titleText = nil
local loginPanel = nil

-- 查找UI组件的辅助函数
-- 支持通过路径查找，例如 "LoginPanel/UsernameInput"
local function FindComponentByPath(path, componentType)
    local go = GameObject.Find(path)
    if go then
        -- 尝试通过字符串类型名获取组件
        local comp = go:GetComponent(componentType)
        if comp then
            return comp
        end
        -- 如果字符串类型名失败，尝试通过类型获取
        if type(componentType) == "string" then
            -- 尝试使用反射方式（如果支持）
            return go:GetComponent(componentType)
        end
    end
    return nil
end

-- 注意：UI多语言文本已在编辑器中通过 LocalizedText 组件设置
-- 无需在此初始化，LocalizedText 组件会在语言切换时自动更新

-- 初始化UI组件
local function InitUIComponents()
    -- 优先从 UIManager 获取 LoginPanel（跨场景 UI）
    local uiManager = GameObject.Find("UIManager")
    if uiManager then
        local manager = uiManager:GetComponent("UIManager")
        if manager then
            loginPanel = manager:GetUI("LoginPanel")
            if loginPanel then
                Debug.Log("[LoginManager] 从 UIManager 获取到 LoginPanel")
            end
        end
    end
    
    -- 如果 UIManager 中没有，尝试在场景中查找（向后兼容）
    if not loginPanel then
        loginPanel = GameObject.Find("LoginPanel")
        if loginPanel then
            Debug.Log("[LoginManager] 在场景中找到 LoginPanel")
        end
    end
    
    if not loginPanel then
        Debug.LogError("[LoginManager] 未找到 LoginPanel，请确保 UIManager 已初始化或场景中存在名为'LoginPanel'的GameObject")
        return false
    end
    
    -- 查找用户名输入框
    -- 尝试多种可能的路径
    usernameInput = FindComponentByPath("LoginPanel/UsernameInput", "TMP_InputField")
    if not usernameInput then
        usernameInput = FindComponentByPath("UsernameInput", "TMP_InputField")
    end
    if not usernameInput and loginPanel then
        -- 尝试在 LoginPanel 的子对象中查找
        local child = loginPanel.transform:Find("UsernameInput")
        if child then
            usernameInput = child.gameObject:GetComponent("TMP_InputField")
        end
    end
    
    -- 查找密码输入框
    passwordInput = FindComponentByPath("LoginPanel/PasswordInput", "TMP_InputField")
    if not passwordInput then
        passwordInput = FindComponentByPath("PasswordInput", "TMP_InputField")
    end
    if not passwordInput and loginPanel then
        local child = loginPanel.transform:Find("PasswordInput")
        if child then
            passwordInput = child.gameObject:GetComponent("TMP_InputField")
        end
    end
    
    -- 查找登录按钮
    local buttonGO = GameObject.Find("LoginPanel/LoginButton")
    if not buttonGO then
        buttonGO = GameObject.Find("LoginButton")
    end
    if not buttonGO and loginPanel then
        local child = loginPanel.transform:Find("LoginButton")
        if child then
            buttonGO = child.gameObject
        end
    end
    if buttonGO then
        loginButton = buttonGO:GetComponent("Button")
    end
    
    -- 查找取消按钮
    local cancelButtonGO = GameObject.Find("LoginPanel/CancelButton")
    if not cancelButtonGO then
        cancelButtonGO = GameObject.Find("CancelButton")
    end
    if not cancelButtonGO and loginPanel then
        local child = loginPanel.transform:Find("CancelButton")
        if child then
            cancelButtonGO = child.gameObject
        end
    end
    if cancelButtonGO then
        cancelButton = cancelButtonGO:GetComponent("Button")
    end
    
    -- 查找设置按钮
    local settingsButtonGO = GameObject.Find("LoginPanel/SettingsButton")
    if not settingsButtonGO then
        settingsButtonGO = GameObject.Find("SettingsButton")
    end
    if not settingsButtonGO and loginPanel then
        local child = loginPanel.transform:Find("SettingsButton")
        if child then
            settingsButtonGO = child.gameObject
        end
    end
    if settingsButtonGO then
        settingsButton = settingsButtonGO:GetComponent("Button")
    end
    
    -- 查找标题文本（支持多种可能的名称）
    local titleGO = GameObject.Find("LoginPanel/TitleText")
    if not titleGO then
        titleGO = GameObject.Find("TitleText")
    end
    if not titleGO then
        titleGO = GameObject.Find("LoginPanel/StatusText")
    end
    if not titleGO and loginPanel then
        local child = loginPanel.transform:Find("TitleText")
        if not child then
            child = loginPanel.transform:Find("StatusText")
        end
        if child then
            titleGO = child.gameObject
        end
    end
    if titleGO then
        titleText = titleGO:GetComponent("TextMeshProUGUI")
    end
    
    -- 查找用户名标签文本
    local usernameTextGO = GameObject.Find("LoginPanel/UsernameText")
    if not usernameTextGO then
        usernameTextGO = GameObject.Find("UsernameText")
    end
    if not usernameTextGO and loginPanel then
        local child = loginPanel.transform:Find("UsernameText")
        if child then
            usernameTextGO = child.gameObject
        end
    end
    if usernameTextGO then
        usernameText = usernameTextGO:GetComponent("TextMeshProUGUI")
    end
    
    -- 查找密码标签文本
    local passwordTextGO = GameObject.Find("LoginPanel/PasswordText")
    if not passwordTextGO then
        passwordTextGO = GameObject.Find("PasswordText")
    end
    if not passwordTextGO and loginPanel then
        local child = loginPanel.transform:Find("PasswordText")
        if child then
            passwordTextGO = child.gameObject
        end
    end
    if passwordTextGO then
        passwordText = passwordTextGO:GetComponent("TextMeshProUGUI")
    end
    
    -- 验证所有组件是否找到
    local missingComponents = {}
    if not usernameInput then table.insert(missingComponents, "UsernameInput") end
    if not passwordInput then table.insert(missingComponents, "PasswordInput") end
    if not loginButton then table.insert(missingComponents, "LoginButton") end
    if not cancelButton then table.insert(missingComponents, "CancelButton") end
    if not settingsButton then table.insert(missingComponents, "SettingsButton") end
    if not titleText then table.insert(missingComponents, "TitleText") end
    if not usernameText then table.insert(missingComponents, "UsernameText") end
    if not passwordText then table.insert(missingComponents, "PasswordText") end
    
    if #missingComponents > 0 then
        Debug.LogWarning("[LoginManager] 缺少以下UI组件: " .. table.concat(missingComponents, ", "))
    end
    
    -- 注意：UI多语言文本已在编辑器中通过 LocalizedText 组件设置，无需在此初始化
    
    return true
end

-- ========== 辅助函数 ==========
-- 隐藏登录界面的通用函数
local function HideLoginPanelAfterDelay(delay)
    local co = coroutine.create(function()
        Yield(WaitForSeconds(delay))
        
        if not loginPanel then
            loginPanel = GameObject.Find("LoginPanel")
        end
        
        if loginPanel then
            loginPanel:SetActive(false)
            Debug.Log("[LoginManager] 登录界面已隐藏")
        else
            Debug.LogError("[LoginManager] 无法隐藏登录界面：loginPanel 为 null")
        end
    end)
    coroutine.resume(co)
end

-- 显示登录成功信息
local function ShowLoginSuccess(message)
    if titleText then
        local successMsg = I18n.Get("login_success")
        if message and message ~= "" then
            successMsg = successMsg .. " " .. message
        end
        titleText.text = successMsg
        titleText.color = Color.green
    end
    HideLoginPanelAfterDelay(1.5)
end

-- 显示登录失败信息
local function ShowLoginFailed(message)
    if titleText then
        local failedMsg = I18n.Get("login_failed")
        if message and message ~= "" then
            failedMsg = failedMsg .. " " .. message
        end
        titleText.text = failedMsg
        titleText.color = Color.red
    end
end

-- ========== 登录逻辑函数 ==========
-- 同步登录：直接调用C#的同步验证函数
local function LoginSync(username, password)
    Debug.Log("[LoginManager] 使用同步登录模式，用户: " .. username)
    
    if not CSharpBridge or not CSharpBridge.LoginService then
        Debug.LogWarning("[LoginManager] CSharpBridge 未初始化，回退到简单验证")
        return LoginSimple(username, password)
    end
    
    local isValid = CSharpBridge.LoginService:ValidateLogin(username, password)
    
    if isValid then
        -- 调用C#函数获取用户信息
        local userInfo = CSharpBridge.LoginService:GetUserInfo(username)
        if userInfo then
            Debug.Log("[LoginManager] 登录成功，用户信息: " .. userInfo)
        end
        
        -- 调用C#函数保存登录状态
        CSharpBridge.LoginService:SaveLoginState(username)
        
        ShowLoginSuccess()
        return true
    else
        Debug.LogWarning("[LoginManager] 登录验证失败，用户名或密码错误")
        ShowLoginFailed(I18n.Get("login_error_username_or_password"))
        return false
    end
end

-- 异步登录：调用C#的异步登录函数，使用回调处理结果
local function LoginAsync(username, password)
    Debug.Log("[LoginManager] 使用异步登录模式，用户: " .. username)
    if not CSharpBridge or not CSharpBridge.LoginService then
        Debug.LogWarning("[LoginManager] CSharpBridge 未初始化，回退到简单验证")
        return LoginSimple(username, password)
    end

    -- 定义Lua回调函数
    local function onLoginSuccess(user, msg)
        Debug.Log("[LoginManager] 异步登录成功，用户: " .. user .. (msg and (", " .. msg) or ""))

        -- 保存登录状态
        CSharpBridge.LoginService:SaveLoginState(user)
        
        -- 显示成功信息并隐藏界面
        ShowLoginSuccess(msg)
    end
    
    local function onLoginFailed(errorMsg)
        Debug.LogWarning("[LoginManager] 异步登录失败: " .. (errorMsg or "未知错误"))
        ShowLoginFailed(errorMsg)
    end
    
    -- 调用C#的异步登录函数，传入Lua回调
    CSharpBridge.LoginService:LoginAsync(username, password, onLoginSuccess, onLoginFailed)
    return true  -- 异步调用立即返回
end

-- 简单验证：向后兼容的简单验证逻辑（不依赖C#）
local function LoginSimple(username, password)
    Debug.LogWarning("[LoginManager] 使用简单验证逻辑（向后兼容模式）")
    
    if username == "admin" and password == "123456" then
        Debug.Log("[LoginManager] 简单验证成功")
        ShowLoginSuccess()
        return true
    else
        Debug.LogWarning("[LoginManager] 简单验证失败，用户名或密码错误")
        ShowLoginFailed()
        return false
    end
end

-- 登录按钮点击处理函数
-- 这个函数展示了如何从Lua调用C#的函数
local function OnLoginButtonClicked()
    if not usernameInput or not passwordInput or not titleText then
        Debug.LogError("[LoginManager] UI 组件未就绪，无法执行登录")
        return
    end
    
    local username = usernameInput.text
    local password = passwordInput.text
    
    Debug.Log("[LoginManager] 登录按钮被点击，用户: " .. username)
    
    -- 简单的验证逻辑
    if not username or username == "" then
        Debug.LogWarning("[LoginManager] 登录失败：用户名为空")
        titleText.text = I18n.Get("login_error_username_empty")
        return
    end
    
    if not password or password == "" then
        Debug.LogWarning("[LoginManager] 登录失败：密码为空")
        titleText.text = I18n.Get("login_error_password_empty")
        return
    end
    
    -- ========== 根据配置变量选择同步或异步登录 ==========
    if LoginManager.UseAsyncLogin then
        -- 使用异步登录
        LoginAsync(username, password)
    else
        -- 使用同步登录
        LoginSync(username, password)
    end
end

-- 取消按钮点击处理函数
local function OnCancelButtonClicked()
    Debug.Log("[LoginManager] 取消按钮被点击")
    
    -- 隐藏登录界面
    if loginPanel then
        loginPanel:SetActive(false)
    else
        Debug.LogWarning("[LoginManager] 无法隐藏登录界面：loginPanel 为 null")
    end
    
    -- 清空输入框
    if usernameInput then
        usernameInput.text = ""
    end
    if passwordInput then
        passwordInput.text = ""
    end
    
    -- 重置标题文本
    if titleText then
        titleText.text = I18n.Get("login_title")
        if Color and Color.white then
            titleText.color = Color.white  -- 重置颜色
        end
    end
end

-- 设置按钮点击处理函数
local function OnSettingsButtonClicked()
    Debug.Log("[LoginManager] 设置按钮被点击")
    
    -- 打开登录面板
    -- 优先通过 UIManager 打开（如果使用跨场景 UI）
    local uiManager = GameObject.Find("UIManager")
    if uiManager then
        local manager = uiManager:GetComponent("UIManager")
        if manager then
            manager:ShowLoginPanel()
            return
        end
    end
    
    -- 如果 UIManager 不存在，直接显示 loginPanel
    if loginPanel then
        loginPanel:SetActive(true)
    else
        Debug.LogWarning("[LoginManager] 无法打开登录面板：loginPanel 为 null")
    end
end

-- ========== 公开的配置和工具函数 ==========
-- 设置登录方式（同步或异步）
-- @param useAsync true=异步登录, false=同步登录
function LoginManager.SetLoginMode(useAsync)
    LoginManager.UseAsyncLogin = useAsync
    local modeText = useAsync and "异步" or "同步"
    Debug.Log("[LoginManager] 登录模式已设置为: " .. modeText)
end

-- 获取当前登录方式
-- @return true=异步, false=同步
function LoginManager.GetLoginMode()
    return LoginManager.UseAsyncLogin
end

-- ========== 多语言相关函数 ==========
-- 切换语言
-- @param langCode 语言代码（使用 BCP 47 标准，如 "zh-CN", "en-US"）
function LoginManager.SetLanguage(langCode)
    if I18n.SetLanguage(langCode) then
        -- LocalizedText 组件会自动更新，无需手动刷新
        Debug.Log("[LoginManager] 语言已切换为: " .. langCode)
        return true
    end
    Debug.LogWarning("[LoginManager] 语言切换失败: " .. langCode)
    return false
end

-- 获取当前语言
function LoginManager.GetCurrentLanguage()
    return I18n.GetCurrentLanguage()
end

-- ========== 供C#调用的Lua函数 ==========
-- 这个函数可以从C#中调用，例如：LoginBridge.CallLuaLoginFunction(username, password)
function LoginManager.LoginFromCSharp(username, password)
    Debug.Log("[LoginManager] 收到来自 C# 的登录请求，用户: " .. username)
    
    -- 可以在这里处理来自C#的登录请求
    if usernameInput and passwordInput then
        usernameInput.text = username
        passwordInput.text = password
    end
    
    -- 调用登录处理函数
    OnLoginButtonClicked()
end

-- 注册Lua回调函数到C#（供C#调用）
function LoginManager.RegisterCallbacksToCSharp()
    if CSharpBridge and CSharpBridge.Bridge then
        -- 定义Lua回调函数
        local function onLoginSuccess(user, msg)
            Debug.Log("[LoginManager] 登录成功回调，用户: " .. user .. (msg and (", " .. msg) or ""))
            if titleText then
                titleText.text = I18n.Get("login_success") .. " " .. (msg or "")
                titleText.color = Color.green
            end
        end
        
        local function onLoginFailed(errorMsg)
            Debug.LogWarning("[LoginManager] 登录失败回调: " .. (errorMsg or "未知错误"))
            if titleText then
                titleText.text = I18n.Get("login_failed") .. " " .. (errorMsg or "")
                titleText.color = Color.red
            end
        end
        
        local function onLoginComplete(success, msg)
            if success then
                Debug.Log("[LoginManager] 登录完成，成功")
            else
                Debug.LogWarning("[LoginManager] 登录完成，失败: " .. (msg or ""))
            end
        end
        
        -- 注册到C#的LoginBridge
        CSharpBridge.Bridge:RegisterLuaLoginCallbacks(onLoginSuccess, onLoginFailed, onLoginComplete)
        Debug.Log("[LoginManager] 回调函数已注册到 C#")
    else
        Debug.LogWarning("[LoginManager] CSharpBridge 未初始化，无法注册回调")
    end
end

-- 初始化函数（相当于 C# 的 Start 方法）
function LoginManager.Start()
    -- 初始化多语言系统
    if not I18n.IsInitialized() then
        I18n.Init()
    end
    
    -- 初始化UI组件
    if InitUIComponents() then
        -- 给登录按钮添加点击事件
        if loginButton then
            loginButton.onClick:AddListener(OnLoginButtonClicked)
            Debug.Log("登录按钮点击事件已绑定")
        end
        
        -- 给取消按钮添加点击事件
        if cancelButton then
            cancelButton.onClick:AddListener(OnCancelButtonClicked)
            Debug.Log("取消按钮点击事件已绑定")
        end
        
        -- 给设置按钮添加点击事件
        if settingsButton then
            settingsButton.onClick:AddListener(OnSettingsButtonClicked)
            Debug.Log("设置按钮点击事件已绑定")
        end
        
        -- 尝试注册回调函数到C#（如果C#桥接已初始化）
        if CSharpBridge and CSharpBridge.Bridge then
            LoginManager.RegisterCallbacksToCSharp()
        end
        
        -- 尝试从C#获取上次登录的用户名
        if CSharpBridge and CSharpBridge.LoginService then
            local lastUser = CSharpBridge.LoginService:GetLastLoginUser()
            if lastUser and lastUser ~= "" and usernameInput then
                usernameInput.text = lastUser
                Debug.Log("[LoginManager] 已填充上次登录的用户名: " .. lastUser)
            end
        end
        
        Debug.Log("[LoginManager] 初始化完成，登录界面已加载")
    else
        Debug.LogError("[LoginManager] 初始化失败：无法找到必要的UI组件")
    end
end

-- 清理函数（相当于 C# 的 OnDestroy 方法）
function LoginManager.OnDestroy()
    -- 清理事件监听（如果需要）
    if loginButton then
        loginButton.onClick:RemoveAllListeners()
    end
    if cancelButton then
        cancelButton.onClick:RemoveAllListeners()
    end
    if settingsButton then
        settingsButton.onClick:RemoveAllListeners()
    end
    
    -- 清空引用
    usernameInput = nil
    passwordInput = nil
    loginButton = nil
    cancelButton = nil
    settingsButton = nil
    titleText = nil
    loginPanel = nil
end

-- 主函数，在 LuaStarter 中调用
function main()
    -- 延迟启动 LoginManager，确保场景完全加载
    -- 使用协程延迟一帧执行
    local co = coroutine.create(function()
        -- 等待一帧，确保C#桥接已初始化
        Yield(null)
        
        -- 初始化多语言系统（优先初始化）
        if not I18n.IsInitialized() then
            I18n.Init()
        end
        
        -- 如果CSharpBridge还未初始化，再等待几帧
        local waitCount = 0
        while (not CSharpBridge or not CSharpBridge.Bridge) and waitCount < 10 do
            Yield(null)
            waitCount = waitCount + 1
        end
        
        -- 启动 LoginManager
        LoginManager.Start()
    end)
    coroutine.resume(co)
end

-- 也可以直接调用（如果场景已经加载完成）
-- 例如从 C# 脚本中调用：LuaSvr.mainState.doString("require 'business.login_ui'; LoginManager.Start()")
function LoginManager.Init()
    LoginManager.Start()
end

return LoginManager
